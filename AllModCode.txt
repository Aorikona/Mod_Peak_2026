		public static void ResetAllFlags()
		{
			var lc = Character.localCharacter;
			if (lc != null)
			{
				if (_mushroomCoroutine != null) { lc.StopCoroutine(_mushroomCoroutine); _mushroomCoroutine = null; }
				if (_greatEpeeEffectsCoroutine != null) { lc.StopCoroutine(_greatEpeeEffectsCoroutine); _greatEpeeEffectsCoroutine = null; }
				if (_greatEpeeStaminaCoroutine != null) { lc.StopCoroutine(_greatEpeeStaminaCoroutine); _greatEpeeStaminaCoroutine = null; }
				RemoveFasterBoi(lc);
				lc.transform.localScale = Vector3.one;
			}
			_greatEpeeNeedsRecovery = false;
			PeakClassModPlugin.scoutActive = false; PeakClassModPlugin.medicActive = false; PeakClassModPlugin.techPriestActive = false;
			PeakClassModPlugin.mushroomancerActive = false; PeakClassModPlugin.greatEpeeActive = false; PeakClassModPlugin.rogueActive = false;
			PeakClassModPlugin.generalActive = false; PeakClassModPlugin.astronautActive = false; PeakClassModPlugin.gobelinActive = false;
			PeakClassModPlugin.wolfActive = false;
			PeakClassModPlugin.currentClass = "None";
		}

		public static void ActivateScoutClass() { ResetAllFlags(); PeakClassModPlugin.scoutActive = true; PeakClassModPlugin.currentClass = "Scout"; }
		public static void ActivateMedicClass() { ResetAllFlags(); PeakClassModPlugin.medicActive = true; PeakClassModPlugin.currentClass = "Medic"; }

		private static readonly string[] ZeroStaminaFungi = new[] { "ShelfShroom", "BounceShroom", "CloudFungus" };

		public static void StartMushroomancerWatch(Character character)
		{
			if (character != null && _mushroomCoroutine == null)
				_mushroomCoroutine = character.StartCoroutine(MushroomancerLocalStaminaWatch(character));
		}

		public static void StartGreatEpeeRoutines(Character character)
		{
			if (character != null)
			{
				if (_greatEpeeEffectsCoroutine == null)
					_greatEpeeEffectsCoroutine = character.StartCoroutine(GreatEpeePermanentEffects(character));
				if (_greatEpeeStaminaCoroutine == null)
					_greatEpeeStaminaCoroutine = character.StartCoroutine(GreatEpeeStaminaRoutine(character));
			}
		}

		public static IEnumerator MushroomancerLocalStaminaWatch(Character character)
		{
			yield return new WaitForSeconds(6f);
			bool wasAboveThreshold = false;

			while (character != null && PeakClassModPlugin.mushroomancerActive && character.IsLocal)
			{
				bool isKnockedOut = character.data != null && (character.data.passedOut || character.data.fullyPassedOut || character.data.dead);
				float currentStam = character.data != null ? character.data.currentStamina : 1f;

				if (currentStam > 0.70f)
				{
					wasAboveThreshold = true;
				}

				if (!isKnockedOut && wasAboveThreshold && currentStam <= 0.03f)
				{
					wasAboveThreshold = false;
					TryActivateStaminaZeroMechanic(character);
				}

				yield return new WaitForSeconds(0.1f);
			}
		}

		public static void TryActivateStaminaZeroMechanic(Character character)
		{
			if (character == null) return;
			int actor = character.photonView?.Owner != null ? character.photonView.Owner.ActorNumber : 0;

			if (lastStaminaZeroTime.TryGetValue(actor, out var last) && Time.time - last < 2.5f) return;
			lastStaminaZeroTime[actor] = Time.time;

			// Application garantie de la malÃ©diction Ã  chaque Ã©puisement d'endurance
			character.refs?.afflictions?.AddStatus(CharacterAfflictions.STATUSTYPE.Curse, ClassBalance.MUSHROOM_CURSE_AMOUNT, false);

			string randomName = ZeroStaminaFungi[UnityEngine.Random.Range(0, ZeroStaminaFungi.Length)];
			Item fungusItem = ItemCache.Get(randomName);
			if (fungusItem == null) return;
			ushort id = (ushort)fungusItem.itemID;

			character.StartCoroutine(MushroomancerGiveItemDelayed(character, id));
		}

		private static IEnumerator MushroomancerGiveItemDelayed(Character character, ushort itemId)
		{
			float timeout = 4f;
			while (character != null && character.data != null && (!character.data.isGrounded || character.data.isSprinting) && timeout > 0)
			{
				timeout -= 0.1f;
				yield return new WaitForSeconds(0.1f);
			}

			if (character?.player != null && character.player.itemSlots != null)
			{
				bool hasSpaceInBaseSlots = false;
				int maxSlotsToCheck = Mathf.Min(3, character.player.itemSlots.Length);

				for (int i = 0; i < maxSlotsToCheck; i++)
				{
					if (character.player.itemSlots[i] != null && character.player.itemSlots[i].IsEmpty())
					{
						hasSpaceInBaseSlots = true;
						break;
					}
				}

				if (hasSpaceInBaseSlots && character.player.HasEmptySlot(itemId))
				{
					character.player.AddItem(itemId, null, out _);
				}
			}
		}

		public static void RemoveFasterBoi(Character character)
		{
			try
			{
				var list = character.refs?.afflictions?.afflictionList;
				if (list != null)
				{
					for (int i = list.Count - 1; i >= 0; i--)
					{
						if (list[i] != null && list[i].GetType().Name.Contains("FasterBoi"))
						{
							list.RemoveAt(i);
						}
					}
				}
			}
			catch { }
		}

		public static IEnumerator GreatEpeePermanentEffects(Character character)
		{
			while (PeakClassModPlugin.greatEpeeActive && character != null && character.IsLocal)
			{
				float currentDrowsy = character.refs?.afflictions?.GetCurrentStatus(CharacterAfflictions.STATUSTYPE.Drowsy) ?? 0f;

				bool isRestricted = _greatEpeeNeedsRecovery
					|| (character.data != null && (character.data.passedOut || character.data.fullyPassedOut))
					|| currentDrowsy > 0.20f;

				if (!isRestricted)
				{
					bool hasFaster = false;
					var list = character.refs?.afflictions?.afflictionList;
					if (list != null)
					{
						for (int i = 0; i < list.Count; i++)
						{
							if (list[i] != null && list[i].GetType().Name.Contains("FasterBoi"))
							{
								hasFaster = true;
								break;
							}
						}
					}

					if (!hasFaster && character.refs?.afflictions != null)
					{
						character.refs.afflictions.AddAffliction(new Peak.Afflictions.Affliction_FasterBoi
						{
							moveSpeedMod = 0.5f,
							climbSpeedMod = 1.0f,
							totalTime = ClassBalance.GREATEPEE_EFFECT_DURATION
						}, false);
					}
				}
				else
				{
					RemoveFasterBoi(character);
				}

				yield return new WaitForSeconds(0.4f);
			}
		}

		public static IEnumerator GreatEpeeStaminaRoutine(Character character)
		{
			yield return new WaitForSeconds(5f);

			while (PeakClassModPlugin.greatEpeeActive && character != null && character.IsLocal)
			{
				float sleepDelay = UnityEngine.Random.Range(ClassBalance.GREATEPEE_SLEEP_MIN, ClassBalance.GREATEPEE_SLEEP_MAX);
				yield return new WaitForSeconds(sleepDelay);

				if (!PeakClassModPlugin.greatEpeeActive || character == null || character.refs?.afflictions == null)
					yield break;

				_greatEpeeNeedsRecovery = true;
				RemoveFasterBoi(character);

				character.refs.afflictions.AddStatus(CharacterAfflictions.STATUSTYPE.Drowsy, 1f, false);

				yield return new WaitForSeconds(0.5f);
				if (character != null && character.data != null && !character.data.dead)
				{
					character.data.passedOut = true;
					character.data.fullyPassedOut = true;
				}

				while (character != null && character.data != null && (character.data.passedOut || character.data.fullyPassedOut))
				{
					yield return new WaitForSeconds(0.5f);
				}

				while (character != null && character.refs?.afflictions != null && character.refs.afflictions.GetCurrentStatus(CharacterAfflictions.STATUSTYPE.Drowsy) > 0.20f)
				{
					yield return new WaitForSeconds(0.2f);
				}

				_greatEpeeNeedsRecovery = false;
				yield return new WaitForSeconds(1f);
			}
		}
	}

using UnityEngine;
using HarmonyLib;
using System.Collections;
using Peak.Afflictions;

namespace PeakClassMod.Handlers
{
    public class AstronautStatHandler : MonoBehaviour
    {
        private Character _character; private CharacterMovement _movement; private Traverse _travMove;
        private float _defJump = 400f; private float _defSprintMult = 2.0f; private bool _defaultsCaptured = false; private bool _isModified = false;
        private Traverse _jumpImpulseField; private Traverse _sprintMultiplierField; private Traverse _velocityField; private bool _fieldsInitialized = false;

        private void Start() { _character = GetComponent<Character>(); _movement = GetComponent<CharacterMovement>(); if (_movement != null) { _travMove = Traverse.Create(_movement); InitializeFields(); } }
        private void InitializeFields() { if (_fieldsInitialized || _travMove == null) return; _jumpImpulseField = _travMove.Field("jumpImpulse"); _sprintMultiplierField = _travMove.Field("sprintMultiplier"); _velocityField = _travMove.Field("velocity"); _fieldsInitialized = true; }
        private void Update() { if (_character == null || !_character.IsLocal || !ClassManager.IsAstronautActive()) return; if (_character.input.jumpWasPressed && _character.data.isGrounded) { StopAllCoroutines(); StartCoroutine(ApplyWindWalkDelayed()); } }
        private IEnumerator ApplyWindWalkDelayed() { yield return new WaitForSeconds(1f); if (ClassManager.IsAstronautActive() && _character != null && !_character.data.isGrounded) TriggerWindWalk(); }
        private void TriggerWindWalk() { try { _character.refs?.afflictions?.AddAffliction(new Affliction_LowGravity(3, ClassBalance.ASTRONAUT_LOW_GRAV_DURATION), true); } catch { } }
        private void FixedUpdate()
        {
            if (_character == null || !_character.IsLocal || !_fieldsInitialized) return;
            if (ClassManager.IsAstronautActive()) { if (!_defaultsCaptured && !_isModified) { CaptureDefaults(); } ApplyAstronautStats(); _isModified = true; }
            else if (_isModified) { RestoreDefaults(); }
        }
        private void CaptureDefaults() { float? j = _jumpImpulseField?.GetValue<float>(); if (j.HasValue && j.Value > 0) _defJump = j.Value; float? sm = _sprintMultiplierField?.GetValue<float>(); if (sm.HasValue) _defSprintMult = sm.Value; _defaultsCaptured = true; }
        private void ApplyAstronautStats()
        {
            _jumpImpulseField?.SetValue(ClassBalance.ASTRONAUT_JUMP_IMPULSE); _sprintMultiplierField?.SetValue(ClassBalance.ASTRONAUT_SPRINT_MULT);
            if (_character.data.isGrounded) { var list = _character.refs?.afflictions?.afflictionList; if (list != null) for (int i = list.Count - 1; i >= 0; i--) if (list[i] is Affliction_LowGravity) _character.refs.afflictions.RemoveAffliction(list[i]); }
            if (!_character.data.isSprinting && _velocityField != null) { Vector3 currentVel = _velocityField.GetValue<Vector3>(); Vector3 hVel = new Vector3(currentVel.x, 0, currentVel.z); if (hVel.magnitude > ClassBalance.ASTRONAUT_MAX_WALK_SPEED) { Vector3 c = hVel.normalized * ClassBalance.ASTRONAUT_MAX_WALK_SPEED; _velocityField.SetValue(new Vector3(c.x, currentVel.y, c.z)); } }
        }
        private void RestoreDefaults() { _jumpImpulseField?.SetValue(_defJump); _sprintMultiplierField?.SetValue(_defSprintMult); _isModified = false; _defaultsCaptured = false; }
    }
}

		private IEnumerator GiveItemsRoutine()
		{
			if (_hasGivenItemsForThisInstance) yield break;

			yield return new WaitForSeconds(1.0f);

			if (_character == null) _character = GetComponent<Character>();

			float timeout = 10f;
			while (timeout > 0)

				if (_character?.player != null && _character.player.itemSlots != null && _character.player.itemSlots.Length > 0)

				timeout -= 0.1f;
				yield return new WaitForSeconds(0.1f);

			if (_hasGivenItemsForThisInstance) yield break;
			_hasGivenItemsForThisInstance = true;

			string myHatName = "None";
			if (HatUtils.TryGetHatName(_character, out string d)) myHatName = d;

			GiveItemsLocally(myHatName);

		{

			// Touche F8 pour purger le slot 0 (tempFullSlot) en cas de besoin

		public void GiveItemsLocally(string hatName)
		{
			if (_character?.player == null) return;

			// Nettoyage prÃ©ventif du slot 0
			if (_character.player.tempFullSlot != null && !_character.player.tempFullSlot.IsEmpty())
			{
				_character.player.tempFullSlot.EmptyOut();
			}

			var itemsToGive = PeakClassModPlugin.BuildStartItemsByHatName(hatName);
			if (itemsToGive == null || itemsToGive.Count == 0) return;

			// Compte le nombre d'exemplaires demandÃ©s par item
			var requestedCounts = new Dictionary<string, int>(StringComparer.OrdinalIgnoreCase);
			foreach (var name in itemsToGive)
			{
				if (!requestedCounts.ContainsKey(name)) requestedCounts[name] = 0;
				requestedCounts[name]++;

				string itemName = kvp.Key;

				var item = ItemCache.Get(itemName);

				ushort id = (ushort)item.itemID;
				int currentCount = CountItemInInventory(_character, id, itemName);
				int needed = targetCount - currentCount;

				for (int i = 0; i < needed; i++)
				{
					// VÃ©rifie qu'il y a au moins un slot d'inventaire de base disponible (Slots 1, 2, 3)
					bool hasBaseSlot = false;

							if (_character.player.itemSlots[s] != null && _character.player.itemSlots[s].IsEmpty())

					}
				}
			}

			// S'assure que le slot 0 n'a pas Ã©tÃ© rempli
			if (_character.player.tempFullSlot != null && !_character.player.tempFullSlot.IsEmpty())

				_character.player.tempFullSlot.EmptyOut();
			}
		}

		public static int CountItemInInventory(Character ch, ushort itemID, string itemName)
		{
			if (ch == null || ch.player == null) return 0;
			int count = 0;

			// 1. Objet actuellement tenu en main
			if (ch.data != null && ch.data.currentItem != null)
			{
				if (ch.data.currentItem.itemID == itemID || ch.data.currentItem.name.IndexOf(itemName, StringComparison.OrdinalIgnoreCase) >= 0)
					count++;
			}

			// 2. Inspection des slots de joueur
			if (ch.player.itemSlots != null)
			{
				foreach (var slot in ch.player.itemSlots)
				{
					if (slot != null && !slot.IsEmpty())
					{
						if (slot.prefab != null && (slot.prefab.itemID == itemID || slot.prefab.name.IndexOf(itemName, StringComparison.OrdinalIgnoreCase) >= 0))
						{
							count++;
							continue;
						}
						string pName = slot.GetPrefabName();
						if (!string.IsNullOrEmpty(pName) && pName.IndexOf(itemName, StringComparison.OrdinalIgnoreCase) >= 0)
						{
							count++;
						}
					}
				}
			}

			// 3. Inspection du sac Ã  dos
			if (ch.player.backpackSlot != null && !ch.player.backpackSlot.IsEmpty())
			{
				if (ch.player.backpackSlot.prefab != null && (ch.player.backpackSlot.prefab.itemID == itemID || ch.player.backpackSlot.prefab.name.IndexOf(itemName, StringComparison.OrdinalIgnoreCase) >= 0))
				{
					count++;
				}
				else
				{
					string bpName = ch.player.backpackSlot.GetPrefabName();
					if (!string.IsNullOrEmpty(bpName) && bpName.IndexOf(itemName, StringComparison.OrdinalIgnoreCase) >= 0)
						count++;
				}
			}

			return count;
		}

		public static bool HasItemInInventory(Character ch, ushort itemID, string itemName)
		{
			return CountItemInInventory(ch, itemID, itemName) > 0;
		}

namespace PeakClassMod.Handlers

	public class ClassWatcherHandler : MonoBehaviour
	{
		private Character _character;
		private string _activeClass = "None";
		private float _lastCheckTime = 0f;
		private float _lastBroadcastTime = 0f;

		private void Start()
		{
			_character = GetComponent<Character>();
			CheckAndApplyClass(true);
		}

		private void Update()
		{
			if (_character == null || !_character.IsLocal) return;

			if (Time.time - _lastCheckTime > 0.5f)
			{
				_lastCheckTime = Time.time;
				CheckAndApplyClass(false);
			}

			// Broadcast pÃ©riodique toutes les 2s pour s'assurer que tous les clients synchronisent la taille
			if (Time.time - _lastBroadcastTime > 2.0f && _activeClass != "None")
			{
				_lastBroadcastTime = Time.time;
				GetComponent<ClassModNetwork>()?.BroadcastClass(_activeClass);
			}
		}

		public void CheckAndApplyClass(bool force)
		{
			if (_character == null || !_character.IsLocal) return;

			if (HatUtils.TryGetHatName(_character, out string hatName))
			{
				string targetClass = DetermineClassFromHat(hatName);
				if (targetClass != "None" && (force || targetClass != _activeClass))
				{
					_activeClass = targetClass;
					ApplyClass(targetClass);
				}
			}
		}

		private string DetermineClassFromHat(string hatName)
		{
			if (string.IsNullOrEmpty(hatName)) return "None";

			if (hatName.Contains("Scoutmaster")) return "Scout";
			if (hatName.Contains("Medic")) return "Medic";
			if (hatName.Contains("Crab")) return "Tech-Priest";
			if (hatName.Contains("Mushroom")) return "Mushroomancer";
			if (hatName.Contains("Propeller")) return "GreatEpee";
			if (hatName.Contains("NinjaHeadband") || hatName.Contains("Ninja")) return "Rogue";
			if (hatName.Contains("Bandana") || hatName.Contains("Captain")) return "General";
			if (hatName.Contains("Astronaut")) return "Astronaut";
			if (hatName.Contains("Midsummer") || hatName.Contains("Gobelin") || hatName.Contains("Crown")) return "Gobelin";
			if (hatName.Contains("WolfEars") || hatName.Contains("Wolf")) return "Wolf";

			return "None";
		}

		private void ApplyClass(string className)
		{
			PeakClassModPlugin.Log?.LogInfo($"[ClassMod] >>> Activation de la classe : {className} <<<");
			GetComponent<ClassModNetwork>()?.BroadcastClass(className);

			switch (className)
			{
				case "Scout": ClassManager.ActivateScoutClass(); break;
				case "Medic":
					ClassManager.ActivateMedicClass();
					if (_character.refs?.afflictions?.currentStatuses != null)
					{
						int idx = (int)CharacterAfflictions.STATUSTYPE.Curse;
						if (_character.refs.afflictions.currentStatuses[idx] < ClassBalance.MEDIC_CURSE)
							_character.refs.afflictions.currentStatuses[idx] = ClassBalance.MEDIC_CURSE;
					}
					break;
				case "Tech-Priest": ClassManager.ActivateTechPriestClass(); break;
				case "Mushroomancer":
					ClassManager.ActivateMushroomancerClass();
					ClassManager.StartMushroomancerWatch(_character);
					break;
				case "GreatEpee":
					ClassManager.ActivateGreatEpeeClass();
					ClassManager.StartGreatEpeeRoutines(_character);
					break;
				case "Rogue": ClassManager.ActivateRogueClass(); break;
				case "General":
					ClassManager.ActivateGeneralClass();
					if (_character.refs?.afflictions?.currentStatuses != null)
					{
						int idx = (int)CharacterAfflictions.STATUSTYPE.Curse;
						if (_character.refs.afflictions.currentStatuses[idx] < ClassBalance.GENERAL_CURSE_MIN)
							_character.refs.afflictions.currentStatuses[idx] = ClassBalance.GENERAL_CURSE_MIN;
					}
					break;
				case "Astronaut": ClassManager.ActivateAstronautClass(); break;
				case "Gobelin": ClassManager.ActivateGobelinClass(); break;
				case "Wolf": ClassManager.ActivateWolfClass(); break;
				default:
					ClassManager.DeactivateAllClasses();
					break;
			}

			try { GUIManager.instance?.bar?.ChangeBar(); } catch { }
		}
	}

		private Dictionary<Character, float> _lastHitTime = new Dictionary<Character, float>();

		private const float EFFECT_RADIUS = 10f;
		private const float ORB_PULSE_SPEED = 2.5f;
		private const float ORB_PULSE_AMPLITUDE = 0.25f;
		private const float LINE_BASE_WIDTH = 0.25f;

		private GameObject _effectAnchor;
		private GameObject _orbContainer;
		private LineRenderer _ringEquator;
		private LineRenderer _ringVertical1;
		private LineRenderer _ringVertical2;

		private void Start()
		{
			_character = GetComponent<Character>();
			_network = GetComponent<ClassModNetwork>();
			_characterClimbing = GetComponent<CharacterClimbing>();
			_characterMovement = GetComponent<CharacterMovement>();

			if (_characterMovement != null) _baseSprintSpeed = _characterMovement.sprintMultiplier;
			if (_characterClimbing != null) _baseClimbSpeed = _characterClimbing.climbSpeed;

			if (_character != null && _character.IsLocal)
			{
				if (_localInstance != null && _localInstance != this)
				{
					Destroy(this);
					return;
				}
				_localInstance = this;
			}
		}

		private void Update()
		{
			if (_character == null || !_character.IsLocal) return;

			bool isWolf = ClassManager.IsWolfActive() || ClassCheckUtil.IsWolf(_character);

			if (!isWolf)
			{
				SetOrbVisibility(false);
				EnsureCannibalAffliction(false);
				if (_timeTextComponent != null) _timeTextComponent.gameObject.SetActive(false);
				return;
			}

			if (_effectAnchor == null)
			{
				Create3DOrb();
				CreateUI();
			}

			if (_timeTextComponent != null && !_timeTextComponent.gameObject.activeSelf)
			{
				_timeTextComponent.gameObject.SetActive(true);
			}

			float pulse = 1f + Mathf.Sin(Time.time * ORB_PULSE_SPEED) * ORB_PULSE_AMPLITUDE;
			float currentWidth = LINE_BASE_WIDTH * pulse;

			if (_ringEquator != null) _ringEquator.widthMultiplier = currentWidth;
			if (_ringVertical1 != null) _ringVertical1.widthMultiplier = currentWidth * 0.9f;
			if (_ringVertical2 != null) _ringVertical2.widthMultiplier = currentWidth * 0.9f;

			if (_orbContainer != null)
			{
				_orbContainer.transform.Rotate(Vector3.up, 25f * Time.deltaTime, Space.World);
			}

			bool isDay = DaySegmentUtils.IsDay();

			// VISION DE CANNIBALE LA NUIT POUR LE LOUP
			EnsureCannibalAffliction(!isDay && !_character.IsGhost);

			if (_character.IsGhost)
			{
				if (_timeTextComponent != null) _timeTextComponent.text = isDay ? "Day" : "Night";
				SetOrbVisibility(false);
				return;
			}
			else
			{
				SetOrbVisibility(true);
			}

			Color orbColor = isDay ? new Color(0.1f, 1f, 0.3f, 0.9f) : new Color(1f, 0.15f, 0.15f, 0.9f);
			SetOrbColor(orbColor);

			if (Time.time >= _nextEffectTime)
			{
				_nextEffectTime = Time.time + 1f;

				if (isDay)
				{
					if (_timeTextComponent != null) _timeTextComponent.text = "Day";
					HealFriends();
					if (_characterClimbing != null) _characterClimbing.climbSpeed = _baseClimbSpeed * ClassBalance.WOLF_CLIMB_SPEED_MULT;
					if (_characterMovement != null) _characterMovement.sprintMultiplier = _baseSprintSpeed;
				}
				else
				{
					if (_timeTextComponent != null) _timeTextComponent.text = "Night";
					DamageAndFeedOnFriends();
					if (_characterClimbing != null) _characterClimbing.climbSpeed = _baseClimbSpeed;
					if (_characterMovement != null) _characterMovement.sprintMultiplier = _baseSprintSpeed * ClassBalance.WOLF_SPRINT_SPEED_MULT;
				}
			}
		}

		private void EnsureCannibalAffliction(bool enable)
		{
			if (_character?.refs?.afflictions == null) return;
			var list = _character.refs.afflictions.afflictionList;
			if (list == null) return;

			if (enable)
			{
				bool hasCannibal = false;
				for (int i = 0; i < list.Count; i++)
				{
					if (list[i] != null && list[i].GetType().Name.IndexOf("Cannibal", StringComparison.OrdinalIgnoreCase) >= 0)
					{
						hasCannibal = true;
						break;
					}
				}

				if (!hasCannibal)
				{
					try
					{
						Type targetType = null;
						foreach (var asm in AppDomain.CurrentDomain.GetAssemblies())
						{
							targetType = asm.GetType("Peak.Afflictions.Affliction_Cannibal")
								?? asm.GetType("Affliction_Cannibal");

							if (targetType == null)
							{
								try
								{
									targetType = asm.GetTypes().FirstOrDefault(t => t != null && t.Name.IndexOf("Cannibal", StringComparison.OrdinalIgnoreCase) >= 0 && typeof(Affliction).IsAssignableFrom(t));
								}
								catch { }
							}

							if (targetType != null) break;
						}

						if (targetType != null)
						{
							Affliction inst = null;
							var ctors = targetType.GetConstructors();
							foreach (var c in ctors)
							{
								var pars = c.GetParameters();
								if (pars.Length == 0) { inst = (Affliction)c.Invoke(null); break; }
								else if (pars.Length == 1 && pars[0].ParameterType == typeof(float)) { inst = (Affliction)c.Invoke(new object[] { 99999f }); break; }
								else if (pars.Length == 2 && pars[0].ParameterType == typeof(int) && pars[1].ParameterType == typeof(float)) { inst = (Affliction)c.Invoke(new object[] { 1, 99999f }); break; }
							}

							if (inst == null)
							{
								inst = (Affliction)System.Runtime.Serialization.FormatterServices.GetUninitializedObject(targetType);
							}

							if (inst != null)
							{
								_character.refs.afflictions.AddAffliction(inst, false);
							}
						}
					}
					catch { }
				}
			}
			else
			{
				for (int i = list.Count - 1; i >= 0; i--)
				{
					if (list[i] != null && list[i].GetType().Name.IndexOf("Cannibal", StringComparison.OrdinalIgnoreCase) >= 0)
					{
						_character.refs.afflictions.RemoveAffliction(list[i]);
					}
				}
			}
		}

		public void DamageAndFeedOnFriends()
		{
			if (_character?.refs?.view == null || !_character.refs.view.IsMine) return;

			Vector3 center = GetRealPlayerPosition();
			Collider[] hitColliders = Physics.OverlapSphere(center, EFFECT_RADIUS);
			HashSet<Character> alreadyHit = new HashSet<Character>();

			bool fedThisTick = false;

			foreach (var hit in hitColliders)
			{
				Character other = hit.GetComponentInParent<Character>();
				if (other == null || other == _character || alreadyHit.Contains(other)) continue;

				alreadyHit.Add(other);
				float now = Time.time;
				if (_lastHitTime.ContainsKey(other) && now - _lastHitTime[other] < 1f) continue;
				_lastHitTime[other] = now;

				var targetNet = other.GetComponent<ClassModNetwork>();
				if (targetNet != null && targetNet.photonView != null && other.photonView != null && other.photonView.Owner != null)
				{
					targetNet.photonView.RPC("RPC_ApplyWolfDamage", other.photonView.Owner, 0.05f);
				}
				else
				{
					other.refs?.afflictions?.AddStatus(CharacterAfflictions.STATUSTYPE.Injury, 0.05f, false, true, true);
				}

				fedThisTick = true;
			}

			// Quand le loup blesse un allié la nuit, il se nourrit et réduit sa faim !
			if (fedThisTick && _character.refs?.afflictions != null)
			{
				_character.refs.afflictions.SubtractStatus(CharacterAfflictions.STATUSTYPE.Hunger, 0.10f, false);
			}
		}

		public void HealFriends()
		{
			if (_character?.refs?.view == null || !_character.refs.view.IsMine) return;

			Vector3 center = GetRealPlayerPosition();
			Collider[] hitColliders = Physics.OverlapSphere(center, EFFECT_RADIUS);
			HashSet<Character> alreadyHit = new HashSet<Character>();

			foreach (var hit in hitColliders)
			{
				Character other = hit.GetComponentInParent<Character>();
				if (other == null || other == _character || alreadyHit.Contains(other)) continue;

				alreadyHit.Add(other);
				float now = Time.time;
				if (_lastHitTime.ContainsKey(other) && now - _lastHitTime[other] < 1f) continue;
				_lastHitTime[other] = now;

				var targetNet = other.GetComponent<ClassModNetwork>();
				if (targetNet != null && targetNet.photonView != null && other.photonView != null && other.photonView.Owner != null)
				{
					targetNet.photonView.RPC("RPC_ApplyWolfHeal", other.photonView.Owner, 0.025f);
				}
				else
				{
					other.refs?.afflictions?.SubtractStatus(CharacterAfflictions.STATUSTYPE.Injury, 0.025f, false);
				}
			}
		}

		private void LateUpdate()
		{
			if (_effectAnchor == null) return;
			_effectAnchor.transform.position = GetRealPlayerPosition();
		}

		private Vector3 GetRealPlayerPosition()
		{
			if (_character != null && _character.IsLocal && Camera.main != null)
			{
				Vector3 cam = Camera.main.transform.position;
				return new Vector3(cam.x, cam.y - 1.0f, cam.z);
			}
			if (_character != null)
			{
				var animPos = _character.transform.Find("animationPosition");
				if (animPos != null) return animPos.position;
				return _character.transform.position;
			}
			return Vector3.zero;
		}

		private void OnDestroy()
		{
			if (_localInstance == this) _localInstance = null;
			EnsureCannibalAffliction(false);
			if (_effectAnchor != null) Destroy(_effectAnchor);
		}

		private void Create3DOrb()
		{
			_effectAnchor = new GameObject("Wolf3DOrbAnchor");
			_effectAnchor.transform.position = GetRealPlayerPosition();

			_orbContainer = new GameObject("OrbRings");
			_orbContainer.transform.SetParent(_effectAnchor.transform, false);
			_orbContainer.transform.localPosition = Vector3.zero;
			_orbContainer.transform.localRotation = Quaternion.identity;

			Shader shader = Shader.Find("Sprites/Default") ?? Shader.Find("GUI/Text Shader") ?? Shader.Find("Unlit/Color");
			Material lineMat = new Material(shader);

			_ringEquator = CreateRing("Equator", _orbContainer.transform, lineMat);
			DrawCircleRing(_ringEquator, EFFECT_RADIUS, Vector3.right, Vector3.forward);

			_ringVertical1 = CreateRing("Meridian1", _orbContainer.transform, lineMat);
			DrawCircleRing(_ringVertical1, EFFECT_RADIUS, Vector3.right, Vector3.up);

			_ringVertical2 = CreateRing("Meridian2", _orbContainer.transform, lineMat);
			DrawCircleRing(_ringVertical2, EFFECT_RADIUS, Vector3.forward, Vector3.up);
		}

		private LineRenderer CreateRing(string name, Transform parent, Material mat)
		{
			GameObject ringObj = new GameObject(name);
			ringObj.transform.SetParent(parent, false);

			LineRenderer lr = ringObj.AddComponent<LineRenderer>();
			lr.useWorldSpace = false;
			lr.loop = true;
			lr.positionCount = 80;
			lr.startWidth = LINE_BASE_WIDTH;
			lr.endWidth = LINE_BASE_WIDTH;
			lr.material = mat;
			lr.shadowCastingMode = UnityEngine.Rendering.ShadowCastingMode.Off;
			lr.receiveShadows = false;
			return lr;
		}

		private void DrawCircleRing(LineRenderer lr, float radius, Vector3 axisA, Vector3 axisB)
		{
			int count = 80;
			for (int i = 0; i < count; i++)
			{
				float angle = ((float)i / count) * Mathf.PI * 2f;
				Vector3 pos = (axisA * Mathf.Cos(angle) + axisB * Mathf.Sin(angle)) * radius;
				lr.SetPosition(i, pos);
			}
		}

		private void SetOrbColor(Color color)
		{
			if (_ringEquator != null) { _ringEquator.startColor = color; _ringEquator.endColor = color; }
			if (_ringVertical1 != null) { _ringVertical1.startColor = color; _ringVertical1.endColor = color; }
			if (_ringVertical2 != null) { _ringVertical2.startColor = color; _ringVertical2.endColor = color; }
		}

		private void SetOrbVisibility(bool visible)
		{
			if (_ringEquator != null) _ringEquator.enabled = visible;
			if (_ringVertical1 != null) _ringVertical1.enabled = visible;
			if (_ringVertical2 != null) _ringVertical2.enabled = visible;
		}

		private void CreateUI()
		{
			try
			{
				GameObject parent = GameObject.Find("GAME/GUIManager/Canvas_HUD");
				if (parent == null && GUIManager.instance != null) parent = GUIManager.instance.gameObject;
				if (parent == null)
				{
					var canvas = UnityEngine.Object.FindObjectsByType<Canvas>(FindObjectsSortMode.None).FirstOrDefault(c => c.renderMode == RenderMode.ScreenSpaceOverlay);
					if (canvas != null) parent = canvas.gameObject;
				}

				if (parent == null) return;

				TMP_FontAsset fontAsset = null;
				foreach (var t in UnityEngine.Object.FindObjectsByType<TextMeshProUGUI>(FindObjectsInactive.Include, FindObjectsSortMode.None))
				{
					if (t.font != null && t.font.name.Contains("Daruma"))
					{
						fontAsset = t.font;
						break;
					}
				}

				GameObject obj = new GameObject("TimeText");
				obj.transform.SetParent(parent.transform, false);

				_timeTextComponent = obj.AddComponent<TextMeshProUGUI>();
				RectTransform rect = obj.GetComponent<RectTransform>();
				rect.anchoredPosition = new Vector2(930, 500);

				_timeTextComponent.fontSize = 40;
				_timeTextComponent.color = Color.white;
				if (fontAsset != null) _timeTextComponent.font = fontAsset;
			}
			catch (Exception ex)
			{
				PeakClassModPlugin.Log?.LogWarning($"[WolfHandler] UI setup warning: {ex.Message}");
			}
		}
	}

		{
			if (__instance?.character == null) return;

			// Poids
			if (statusType == CharacterAfflictions.STATUSTYPE.Weight)
			{
				if (ClassManager.IsTechPriestActive() && ClassCheckUtil.IsTechPriest(__instance.character))
					__result = 0f;
				else if (ClassManager.IsScoutActive() && ClassCheckUtil.IsScout(__instance.character))

				else if (ClassManager.IsGeneralActive() && ClassCheckUtil.IsGeneral(__instance.character))
					__result *= ClassBalance.GENERAL_WEIGHT_MULT;
			}

			// Malédiction
			if (statusType == CharacterAfflictions.STATUSTYPE.Curse)
			{
				if (ClassManager.IsGeneralActive() && ClassCheckUtil.IsGeneral(__instance.character))
				{
					if (__result < ClassBalance.GENERAL_CURSE_MIN) __result = ClassBalance.GENERAL_CURSE_MIN;

			if (statusType == CharacterAfflictions.STATUSTYPE.Poison && (ClassCheckUtil.IsMushroom(__instance.character) || ClassManager.IsMushroomancerActive()))
				__result = 0f;

			if (statusType == CharacterAfflictions.STATUSTYPE.Spores && (ClassCheckUtil.IsGobelin(__instance.character) || ClassManager.IsGobelinActive()))
				__result = 0f;
		}

			{

			}

using System;
using System.Collections.Generic;
using UnityEngine;

namespace PeakClassMod.Utils
{
	public static class GobelinDiet
	{
		// Whitelist explicite d'aliments naturels autorisÃ©s pour le Gobelin
		private static readonly HashSet<string> NaturalItemNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
		{
			"Shroomberry_Red", "Shroomberry_Yellow", "Shroomberry_Purple", "Shroomberry_Blue", "Shroomberry_Green",
			"ShelfShroom", "BounceShroom", "CloudFungus", "HealingPuffShroom", "PoisonShroom",
			"MagicBean", "HealingDart Variant", "HealingDart", "BalloonBunch", "RopeSpool", "ChainShooter"
		};

		private static readonly string[] NaturalKeywords = new[]
		{
			"shroom", "mushroom", "berry", "shroomberry", "insect", "bug", "raw",
			"fish", "meat", "plant", "herb", "puff", "spore", "healingpuff", "bean", "magicbean",
			"worm", "fly", "centipede", "beetle", "grub", "root", "seed", "fern", "flower", "cactus", "aloe", "bird", "egg"
		};

		// Blacklist explicite de nourriture industrielle / transformÃ©e
		private static readonly string[] PackagedKeywords = new[]
		{
			"packaged", "bar", "energybar", "granola", "can", "canned", "chips",
			"cracker", "ration", "soda", "cola", "candy", "chocolate", "cookie",
			"marshmallow", "pie", "drink", "snack", "soup", "tuna", "sardine",
			"noodles", "mre", "bread", "energydrink", "popcorn", "crisps", "pastry",
			"meatpie", "peanuts", "nuts", "jerky", "beef", "cheese", "sausage", "bacon",
			"sandwich", "burger", "hotdog", "pizza", "pasta", "ramen", "cereal", "gum", "cake"
		};

		public static bool IsNaturalFood(Item item)
		{
			if (item == null) return false;

			string rawName = (item.name ?? "").Replace("(Clone)", "").Trim().ToLowerInvariant();
			string uiName = (item.UIData?.itemName ?? "").ToLowerInvariant();

			if (NaturalItemNames.Contains(item.name) || NaturalItemNames.Contains(rawName))
				return true;

			if ((item.itemTags & Item.ItemTags.Berry) != 0 || (item.itemTags & Item.ItemTags.Mushroom) != 0 || (item.itemTags & Item.ItemTags.Bird) != 0)
				return true;

			foreach (var nat in NaturalKeywords)
			{
				if (rawName.Contains(nat) || uiName.Contains(nat))
					return true;
			}

			return false;
		}

		public static bool IsFood(Item item)
		{
			if (item == null) return false;

			if ((item.itemTags & Item.ItemTags.PackagedFood) != 0)
				return true;

			var playAnim = item.GetComponent<Action_PlayAnimation>();
			if (playAnim != null && !string.IsNullOrEmpty(playAnim.animationName))
			{
				string a = playAnim.animationName.ToLowerInvariant();
				if (a.Contains("eat") || a.Contains("drink") || a.Contains("consume")) return true;
			}

			var feedback = item.GetComponent<ItemUseFeedback>();
			if (feedback != null && !string.IsNullOrEmpty(feedback.useAnimation))
			{
				string a = feedback.useAnimation.ToLowerInvariant();
				if (a.Contains("eat") || a.Contains("drink") || a.Contains("consume")) return true;
			}

			var modStatuses = item.GetComponents<Action_ModifyStatus>();
			foreach (var ms in modStatuses)
			{
				if (ms.statusType == CharacterAfflictions.STATUSTYPE.Hunger) return true;
			}

			if (item.GetComponent<Action_RestoreHunger>() != null)
				return true;

			return false;
		}

		public static bool IsPackagedFood(Item item)
		{
			if (item == null) return false;

			// 1. Si c'est un aliment naturel, JAMAIS interdit
			if (IsNaturalFood(item))
				return false;

			// 2. Si c'est de la nourriture consommable et pas naturelle, INTERDIT
			if (IsFood(item))
				return true;

			// 3. Mots-clÃ©s industriels / emballÃ©s
			string rawName = (item.name ?? "").Replace("(Clone)", "").Trim().ToLowerInvariant();
			string uiName = (item.UIData?.itemName ?? "").ToLowerInvariant();

			foreach (var kw in PackagedKeywords)
			{
				if (rawName.Contains(kw) || uiName.Contains(kw))
					return true;
			}

			return false;
		}
	}
}

using System;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using Zorro.Core;
using Photon.Pun;

namespace PeakClassMod.Utils
{
	public static class HatUtils
	{
		private static string _lastLocalHatName = "None";

		public static void SetLastLocalHatName(string hatName)
		{
			if (!string.IsNullOrEmpty(hatName) && hatName != "Unknown" && hatName != "None")
			{
				_lastLocalHatName = hatName;
			}
		}

		public static string GetLastLocalHatName() => _lastLocalHatName;

		public static bool TryGetHatName(Character ch, out string hatName)
		{
			hatName = null;
			if (ch == null) return false;

			// --- SOURCE 1 : Inspection des GameObjects de chapeaux actifs ---
			try
			{
				if (ch.refs?.customization?.refs?.playerHats != null)
				{
					foreach (var h in ch.refs.customization.refs.playerHats)
					{
						if (h != null && h.gameObject.activeInHierarchy)
						{
							hatName = h.gameObject.name;
							if (ch.IsLocal) SetLastLocalHatName(hatName);
							return true;
						}
					}
				}

				var transforms = ch.GetComponentsInChildren<Transform>(true);
				foreach (var t in transforms)
				{
					if (t == null || !t.gameObject.activeInHierarchy) continue;
					string objName = t.name;
					if (objName.StartsWith("Hat_", StringComparison.OrdinalIgnoreCase) || objName.Contains("Crown") || objName.Contains("Helmet") || objName.Contains("Bandana"))
					{
						hatName = objName;
						if (ch.IsLocal) SetLastLocalHatName(hatName);
						return true;
					}
				}
			}
			catch { }

			// --- SOURCE 2 : PersistentPlayerDataService & Customization ---
			try
			{
				var ppds = GameHandler.GetService<PersistentPlayerDataService>();
				PersistentPlayerData data = null;
				if (ch.photonView?.Owner != null) data = ppds?.GetPlayerData(ch.photonView.Owner);
				if (data == null && ch.IsLocal && PhotonNetwork.LocalPlayer != null) data = ppds?.GetPlayerData(PhotonNetwork.LocalPlayer);

				if (data != null && Singleton<Customization>.Instance != null)
				{
					var hats = Singleton<Customization>.Instance.hats;
					int i = data.customizationData.currentHat;
					if (hats != null && i >= 0 && i < hats.Length && hats[i] != null)
					{
						hatName = hats[i].name;
						if (ch.IsLocal) SetLastLocalHatName(hatName);
						return true;
					}
				}
			}
			catch { }

			// --- SOURCE 3 : MÃ©moire locale persistÃ©e ---
			if (ch.IsLocal && !string.IsNullOrEmpty(_lastLocalHatName) && _lastLocalHatName != "None")
			{
				hatName = _lastLocalHatName;
				return true;
			}

			return false;
		}

		public static bool HasAnyHat(Character ch, params string[] names)
		{
			if (!TryGetHatName(ch, out var name)) return false;
			return names.Any(n => name.IndexOf(n, StringComparison.OrdinalIgnoreCase) >= 0);
		}

		public static bool HasHat(Character ch, string expected)
		{
			return TryGetHatName(ch, out var name) && name.IndexOf(expected, StringComparison.OrdinalIgnoreCase) >= 0;
		}
	}
}

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using UnityEngine;
using UnityEngine.Networking;
using System.Collections;

namespace PeakClassMod.Utils
{
    public class ImageLoader : MonoBehaviour
    {
        public static ImageLoader Instance;

        void Awake()
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
        }

        public void Load(string path, Action<Texture2D> callback)
        {
            StartCoroutine(LoadCoroutine(path, callback));
        }

        private IEnumerator LoadCoroutine(string path, Action<Texture2D> callback)
        {
            string url = "file://" + path;

            UnityWebRequest request = UnityWebRequestTexture.GetTexture(url);
            yield return request.SendWebRequest();

            if (request.result != UnityWebRequest.Result.Success)
            {
                Debug.LogError(request.error);
                callback(null);
            }
            else
            {
                callback(DownloadHandlerTexture.GetContent(request));
            }
        }
    }
}

using System;
using System.Collections.Generic;
using UnityEngine;

namespace PeakClassMod.Utils
{
	public static class ItemCache
	{
		private static Dictionary<string, Item> _byName;
		private static float _lastRefreshTime = 0f;

		private static string Normalize(string input)
		{
			if (string.IsNullOrEmpty(input)) return "";
			return input.Replace(" ", "").Replace("-", "").Replace("_", "").Replace("(Clone)", "").Trim().ToLowerInvariant();
		}

		public static void Refresh()
		{
			_byName = new Dictionary<string, Item>(StringComparer.OrdinalIgnoreCase);
			var allItems = Resources.FindObjectsOfTypeAll<Item>();
			foreach (var it in allItems)
			{
				if (it == null || string.IsNullOrEmpty(it.name)) continue;
				string raw = it.name;
				string clean = raw.Replace("(Clone)", "").Trim();
				string norm = Normalize(raw);

				if (!_byName.ContainsKey(raw)) _byName[raw] = it;
				if (!_byName.ContainsKey(clean)) _byName[clean] = it;
				if (!_byName.ContainsKey(norm)) _byName[norm] = it;

				// Log tous les items pour trouver le nom exact
				if (raw.ToLower().Contains("dart") || raw.ToLower().Contains("blow") || raw.ToLower().Contains("heal") || raw.ToLower().Contains("gun") || raw.ToLower().Contains("pipe") || raw.ToLower().Contains("sarbacane"))
				{
					PeakClassModPlugin.Log?.LogInfo($"[ItemScan] Dart/Weapon candidate: '{raw}' (ID: {it.itemID})");
				}
			}
			_lastRefreshTime = Time.time;
		}

		public static Item Get(string name)
		{
			if (string.IsNullOrEmpty(name)) return null;

			if (_byName == null || Time.time - _lastRefreshTime > ClassBalance.ITEM_CACHE_DURATION)
			{
				Refresh();
			}

			if (_byName.TryGetValue(name, out var item)) return item;

			string searchNorm = Normalize(name);
			if (_byName.TryGetValue(searchNorm, out item)) return item;

			// Recherche floue
			foreach (var kvp in _byName)
			{
				string keyNorm = Normalize(kvp.Key);
				if (keyNorm.Contains(searchNorm) || searchNorm.Contains(keyNorm))
				{
					return kvp.Value;
				}
			}

			return null;
		}
	}
}

using System;
using UnityEngine;

namespace PeakClassMod.Utils
{
	internal static class TechPriestBlocker
	{
		private static readonly string[] HealingKeywords = new[]
		{
			"heal", "cure", "medicine", "restorehealth", "bandage", "firstaid",
			"medkit", "salve", "pill", "healingpuff", "cure-all", "cureall"
		};

		public static bool ShouldBlockSelfUse(Item item)
		{
			if (item == null) return false;

			string rawName = (item.name ?? "").ToLowerInvariant();
			string uiName = (item.UIData?.itemName ?? "").ToLowerInvariant();

			// 1. Bloque les items de soin pour soi-mÃªme
			foreach (var word in HealingKeywords)
			{
				if (rawName.Contains(word) || uiName.Contains(word))
					return true;
			}

			// 2. Bloque la nourriture organique / naturelle pour soi-mÃªme
			if (GobelinDiet.IsNaturalFood(item))
			{
				return true;
			}

			// 3. Bloque toute nourriture consommable directement
			if (GobelinDiet.IsFood(item))
			{
				return true;
			}

			// 4. Bloque les boosts d'endurance et boissons Ã©nergisantes
			var affAction = item.GetComponent<Action_ApplyAffliction>();
			if (affAction != null && affAction.affliction != null)
			{
				string affName = affAction.affliction.GetType().Name.ToLowerInvariant();
				if (affName.Contains("fasterboi") || affName.Contains("stamina") || affName.Contains("energy"))
					return true;
			}

			return false;
		}
	}
}
