using System;

	{
		private static readonly Dictionary<int, float> lastStaminaZeroTime = new Dictionary<int, float>();
		private static readonly Dictionary<int, int> staminaZeroCount = new Dictionary<int, int>();
		private static Coroutine _mushroomCoroutine;
		private static Coroutine _greatEpeeEffectsCoroutine;
		private static Coroutine _greatEpeeStaminaCoroutine;
		private static bool _greatEpeeNeedsRecovery = false;

		public static void ResetAllFlags()
		{
			var lc = Character.localCharacter;
			if (lc != null)
			{
				if (_mushroomCoroutine != null) { lc.StopCoroutine(_mushroomCoroutine); _mushroomCoroutine = null; }
				if (_greatEpeeEffectsCoroutine != null) { lc.StopCoroutine(_greatEpeeEffectsCoroutine); _greatEpeeEffectsCoroutine = null; }
				if (_greatEpeeStaminaCoroutine != null) { lc.StopCoroutine(_greatEpeeStaminaCoroutine); _greatEpeeStaminaCoroutine = null; }
				RemoveFasterBoi(lc);
				lc.transform.localScale = Vector3.one;
			}
			_greatEpeeNeedsRecovery = false;
			PeakClassModPlugin.scoutActive = false; PeakClassModPlugin.medicActive = false; PeakClassModPlugin.techPriestActive = false;
			PeakClassModPlugin.mushroomancerActive = false; PeakClassModPlugin.greatEpeeActive = false; PeakClassModPlugin.rogueActive = false;
			PeakClassModPlugin.generalActive = false; PeakClassModPlugin.astronautActive = false; PeakClassModPlugin.gobelinActive = false;
			PeakClassModPlugin.wolfActive = false;
			PeakClassModPlugin.currentClass = "None";
		}

		public static void ActivateScoutClass() { ResetAllFlags(); PeakClassModPlugin.scoutActive = true; PeakClassModPlugin.currentClass = "Scout"; }
		public static void ActivateMedicClass() { ResetAllFlags(); PeakClassModPlugin.medicActive = true; PeakClassModPlugin.currentClass = "Medic"; }
		public static void ActivateTechPriestClass() { ResetAllFlags(); PeakClassModPlugin.techPriestActive = true; PeakClassModPlugin.currentClass = "Tech-Priest"; }

		private static readonly string[] ZeroStaminaFungi = new[] { "ShelfShroom", "BounceShroom", "CloudFungus" };

		public static void StartMushroomancerWatch(Character character)
		{
			if (character != null && _mushroomCoroutine == null)
				_mushroomCoroutine = character.StartCoroutine(MushroomancerLocalStaminaWatch(character));
		}

		public static void StartGreatEpeeRoutines(Character character)
		{
			if (character != null)
			{
				if (_greatEpeeEffectsCoroutine == null)
					_greatEpeeEffectsCoroutine = character.StartCoroutine(GreatEpeePermanentEffects(character));
				if (_greatEpeeStaminaCoroutine == null)
					_greatEpeeStaminaCoroutine = character.StartCoroutine(GreatEpeeStaminaRoutine(character));
			}
		}

		public static IEnumerator MushroomancerLocalStaminaWatch(Character character)
		{
			// Délai initial de 4s pour laisser le joueur charger son endurance à 100%
			yield return new WaitForSeconds(4f);

			bool wasAboveThreshold = true;

			while (character != null && PeakClassModPlugin.mushroomancerActive && character.IsLocal)
			{
				bool isKnockedOut = false;
				if (character.data != null)
				{
					if (character.data.passedOut || character.data.fullyPassedOut || character.data.dead)
						isKnockedOut = true;
				}

				float currentStam = character.data != null ? character.data.currentStamina : 1f;

				if (currentStam > 0.20f)
				{
					wasAboveThreshold = true;
				}

				// Ne donne un champignon QUE si le joueur avait de l'endurance et vient de la vider
				if (!isKnockedOut && wasAboveThreshold && currentStam <= 0.02f)
				{
					wasAboveThreshold = false;
					TryActivateStaminaZeroMechanic(character);
				}

				yield return new WaitForSeconds(0.1f);
			}
		}

		public static void TryActivateStaminaZeroMechanic(Character character)
		{
			if (character?.photonView?.Owner == null) return;
			int actor = character.photonView.Owner.ActorNumber;
			if (lastStaminaZeroTime.TryGetValue(actor, out var last) && Time.time - last < ClassBalance.MUSHROOM_STAMINA_COOLDOWN) return;
			lastStaminaZeroTime[actor] = Time.time;

			if (!staminaZeroCount.ContainsKey(actor)) staminaZeroCount[actor] = 0;
			staminaZeroCount[actor]++;

			if (staminaZeroCount[actor] % 2 == 0)
			{
				character.refs?.afflictions?.AddStatus(CharacterAfflictions.STATUSTYPE.Curse, ClassBalance.MUSHROOM_CURSE_AMOUNT, false);
			}

			string randomName = ZeroStaminaFungi[UnityEngine.Random.Range(0, ZeroStaminaFungi.Length)];
			Item fungusItem = ItemCache.Get(randomName);
			if (fungusItem == null) return;
			ushort id = (ushort)fungusItem.itemID;

			if (character != null)
			{
				character.StartCoroutine(MushroomancerGiveItemDelayed(character, id));
			}
		}

		private static IEnumerator MushroomancerGiveItemDelayed(Character character, ushort itemId)
		{
			float timeout = 5f;
			while (character != null && character.data != null && (!character.data.isGrounded || character.data.isSprinting) && timeout > 0)
			{
				timeout -= 0.1f;
				yield return new WaitForSeconds(0.1f);
			}

			if (character != null && character.player != null && character.player.itemSlots != null)
			{
				bool hasSpaceInBaseSlots = false;
				int maxSlotsToCheck = Mathf.Min(3, character.player.itemSlots.Length);

				for (int i = 0; i < maxSlotsToCheck; i++)
				{
					if (character.player.itemSlots[i] != null && character.player.itemSlots[i].IsEmpty())
					{
						hasSpaceInBaseSlots = true;
						break;
					}
				}

				if (hasSpaceInBaseSlots && character.player.HasEmptySlot(itemId))
				{
					character.player.AddItem(itemId, null, out _);
				}
			}
		}

		public static void RemoveFasterBoi(Character character)
		{
			try
			{
				var list = character.refs?.afflictions?.afflictionList;
				if (list != null)
				{
					for (int i = list.Count - 1; i >= 0; i--)
					{
						if (list[i] != null && list[i].GetType().Name.Contains("FasterBoi"))
						{
							list.RemoveAt(i);
						}
					}
				}
			}
			catch { }
		}

		public static IEnumerator GreatEpeePermanentEffects(Character character)
		{
			while (PeakClassModPlugin.greatEpeeActive && character != null && character.IsLocal)
			{
				float currentDrowsy = character.refs?.afflictions?.GetCurrentStatus(CharacterAfflictions.STATUSTYPE.Drowsy) ?? 0f;

				bool isRestricted = _greatEpeeNeedsRecovery
					|| (character.data != null && (character.data.passedOut || character.data.fullyPassedOut))
					|| currentDrowsy > 0.20f;

				if (!isRestricted)
				{
					bool hasFaster = false;
					var list = character.refs?.afflictions?.afflictionList;
					if (list != null)
					{
						for (int i = 0; i < list.Count; i++)
						{
							if (list[i] != null && list[i].GetType().Name.Contains("FasterBoi"))
							{
								hasFaster = true;
								break;
							}
						}
					}

					if (!hasFaster && character.refs?.afflictions != null)
					{
						character.refs.afflictions.AddAffliction(new Peak.Afflictions.Affliction_FasterBoi
						{
							moveSpeedMod = 0.5f,
							climbSpeedMod = 1.0f,
							totalTime = ClassBalance.GREATEPEE_EFFECT_DURATION
						}, false);
					}
				}
				else
				{
					RemoveFasterBoi(character);
				}

				yield return new WaitForSeconds(0.4f);
			}
		}

		public static IEnumerator GreatEpeeStaminaRoutine(Character character)
		{
			yield return new WaitForSeconds(5f);

			while (PeakClassModPlugin.greatEpeeActive && character != null && character.IsLocal)
			{
				float sleepDelay = UnityEngine.Random.Range(ClassBalance.GREATEPEE_SLEEP_MIN, ClassBalance.GREATEPEE_SLEEP_MAX);
				yield return new WaitForSeconds(sleepDelay);

				if (!PeakClassModPlugin.greatEpeeActive || character == null || character.refs?.afflictions == null)
					yield break;

				_greatEpeeNeedsRecovery = true;
				RemoveFasterBoi(character);

				character.refs.afflictions.AddStatus(CharacterAfflictions.STATUSTYPE.Drowsy, 1f, false);

				yield return new WaitForSeconds(0.5f);
				if (character != null && character.data != null && !character.data.dead)
				{
					character.data.passedOut = true;
					character.data.fullyPassedOut = true;
				}

				while (character != null && character.data != null && (character.data.passedOut || character.data.fullyPassedOut))
				{
					yield return new WaitForSeconds(0.5f);
				}

				while (character != null && character.refs?.afflictions != null && character.refs.afflictions.GetCurrentStatus(CharacterAfflictions.STATUSTYPE.Drowsy) > 0.20f)
				{
					yield return new WaitForSeconds(0.2f);
				}

				_greatEpeeNeedsRecovery = false;
				yield return new WaitForSeconds(1f);
			}
		}
	}
}

		public static List<string> BuildStartItemsByHatName(string hatName)
		{
			var items = new List<string>();
			if (string.IsNullOrEmpty(hatName)) return items;

			if (hatName.IndexOf("Mushroom", StringComparison.OrdinalIgnoreCase) >= 0)
			{
				for (int i = 0; i < 3; i++) items.Add(StartBerries[UnityEngine.Random.Range(0, StartBerries.Length)]);
			}
			else if (hatName.IndexOf("Medic", StringComparison.OrdinalIgnoreCase) >= 0)
			{
				items.Add("FirstAidKit");
				items.Add("Cure-All");
			}
			else if (hatName.IndexOf("Crab", StringComparison.OrdinalIgnoreCase) >= 0)
			{
				items.Add("ChainShooter");
				items.Add("ChainShooter");
			}
			else if (hatName.IndexOf("Ninja", StringComparison.OrdinalIgnoreCase) >= 0)
			{
				items.Add("RopeSpool");
				items.Add("RopeSpool");
			}
			else if (hatName.IndexOf("Astronaut", StringComparison.OrdinalIgnoreCase) >= 0)
			{
				items.Add("BalloonBunch");
			}
			else if (hatName.IndexOf("Midsummer", StringComparison.OrdinalIgnoreCase) >= 0 ||
					 hatName.IndexOf("Gobelin", StringComparison.OrdinalIgnoreCase) >= 0)
			{
				items.Add(ClassBalance.BLOWGUN); // "HealingDart Variant"
				items.Add("MagicBean");
			}

			return items;
		}

using UnityEngine;
using HarmonyLib;
using System.Collections;
using Peak.Afflictions;

namespace PeakClassMod.Handlers
{
    public class AstronautStatHandler : MonoBehaviour
    {
        private Character _character; private CharacterMovement _movement; private Traverse _travMove;
        private float _defJump = 400f; private float _defSprintMult = 2.0f; private bool _defaultsCaptured = false; private bool _isModified = false;
        private Traverse _jumpImpulseField; private Traverse _sprintMultiplierField; private Traverse _velocityField; private bool _fieldsInitialized = false;

        private void Start() { _character = GetComponent<Character>(); _movement = GetComponent<CharacterMovement>(); if (_movement != null) { _travMove = Traverse.Create(_movement); InitializeFields(); } }
        private void InitializeFields() { if (_fieldsInitialized || _travMove == null) return; _jumpImpulseField = _travMove.Field("jumpImpulse"); _sprintMultiplierField = _travMove.Field("sprintMultiplier"); _velocityField = _travMove.Field("velocity"); _fieldsInitialized = true; }
        private void Update() { if (_character == null || !_character.IsLocal || !ClassManager.IsAstronautActive()) return; if (_character.input.jumpWasPressed && _character.data.isGrounded) { StopAllCoroutines(); StartCoroutine(ApplyWindWalkDelayed()); } }
        private IEnumerator ApplyWindWalkDelayed() { yield return new WaitForSeconds(1f); if (ClassManager.IsAstronautActive() && _character != null && !_character.data.isGrounded) TriggerWindWalk(); }
        private void TriggerWindWalk() { try { _character.refs?.afflictions?.AddAffliction(new Affliction_LowGravity(3, ClassBalance.ASTRONAUT_LOW_GRAV_DURATION), true); } catch { } }
        private void FixedUpdate()
        {
            if (_character == null || !_character.IsLocal || !_fieldsInitialized) return;
            if (ClassManager.IsAstronautActive()) { if (!_defaultsCaptured && !_isModified) { CaptureDefaults(); } ApplyAstronautStats(); _isModified = true; }
            else if (_isModified) { RestoreDefaults(); }
        }
        private void CaptureDefaults() { float? j = _jumpImpulseField?.GetValue<float>(); if (j.HasValue && j.Value > 0) _defJump = j.Value; float? sm = _sprintMultiplierField?.GetValue<float>(); if (sm.HasValue) _defSprintMult = sm.Value; _defaultsCaptured = true; }
        private void ApplyAstronautStats()
        {
            _jumpImpulseField?.SetValue(ClassBalance.ASTRONAUT_JUMP_IMPULSE); _sprintMultiplierField?.SetValue(ClassBalance.ASTRONAUT_SPRINT_MULT);
            if (_character.data.isGrounded) { var list = _character.refs?.afflictions?.afflictionList; if (list != null) for (int i = list.Count - 1; i >= 0; i--) if (list[i] is Affliction_LowGravity) _character.refs.afflictions.RemoveAffliction(list[i]); }
            if (!_character.data.isSprinting && _velocityField != null) { Vector3 currentVel = _velocityField.GetValue<Vector3>(); Vector3 hVel = new Vector3(currentVel.x, 0, currentVel.z); if (hVel.magnitude > ClassBalance.ASTRONAUT_MAX_WALK_SPEED) { Vector3 c = hVel.normalized * ClassBalance.ASTRONAUT_MAX_WALK_SPEED; _velocityField.SetValue(new Vector3(c.x, currentVel.y, c.z)); } }
        }
        private void RestoreDefaults() { _jumpImpulseField?.SetValue(_defJump); _sprintMultiplierField?.SetValue(_defSprintMult); _isModified = false; _defaultsCaptured = false; }
    }
}

using Photon.Pun;
using Photon.Realtime;
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using PeakClassMod.Utils;

namespace PeakClassMod.Handlers
{
	public class ClassModNetwork : MonoBehaviourPun
	{
		private Character _character;
		private bool _hasRequestedItems = false;
		private static Dictionary<int, float> _hostGiveCooldown = new Dictionary<int, float>();
		private static Dictionary<int, Character> _characterCache = new Dictionary<int, Character>();
		public static HashSet<int> HasReceivedStartItems = new HashSet<int>();

		private void Awake()
		{
			var c = GetComponents<ClassModNetwork>();
			if (c.Length > 1) { Destroy(this); return; }
			_character = GetComponent<Character>();
		}

		private void Start()
		{
			if (photonView != null && photonView.IsMine && !_hasRequestedItems)
			{
				_hasRequestedItems = true;
				StartCoroutine(RequestItemsRoutine());
			}
		}

		private void OnDestroy()
		{
			if (_character?.photonView?.Owner != null)
				_characterCache.Remove(_character.photonView.Owner.ActorNumber);
		}

		public static void ClearPlayerCache(int actorNumber)
		{
			_characterCache.Remove(actorNumber);
			_hostGiveCooldown.Remove(actorNumber);
			HasReceivedStartItems.Remove(actorNumber);
		}

		private IEnumerator RequestItemsRoutine()
		{
			yield return new WaitForSeconds(ClassBalance.ITEM_REQUEST_DELAY);

			if (_character == null) _character = GetComponent<Character>();

			// Attend que le joueur et ses slots d'inventaire soient initialisés
			float timeout = 6f;
			while (timeout > 0)
			{
				if (_character?.player != null && _character.player.itemSlots != null && _character.player.itemSlots.Length > 0)
				{
					break;
				}
				timeout -= 0.1f;
				yield return new WaitForSeconds(0.1f);
			}

			yield return new WaitForSeconds(0.5f);

			string myHatName = "None";
			if (HatUtils.TryGetHatName(_character, out string d)) myHatName = d;

			if (PhotonNetwork.IsConnected && PhotonNetwork.InRoom)
			{
				photonView.RPC("RPC_HostGiveItems", RpcTarget.MasterClient, myHatName);
			}
			else
			{
				// Mode Solo
				GiveItemsLocally(myHatName);
			}
		}

		public void GiveItemsLocally(string hatName)
		{
			if (_character?.player == null) return;
			var itemsToGive = PeakClassModPlugin.BuildStartItemsByHatName(hatName);
			foreach (var itemName in itemsToGive)
			{
				var item = ItemCache.Get(itemName);
				if (item != null && _character.player.HasEmptySlot((ushort)item.itemID))
				{
					_character.player.AddItem((ushort)item.itemID, null, out _);
				}
			}
		}

		[PunRPC]
		public void RPC_HostGiveItems(string clientHatName, PhotonMessageInfo info)
		{
			if (!PhotonNetwork.IsMasterClient) return;
			int actorID = info.Sender.ActorNumber;
			if (HasReceivedStartItems.Contains(actorID)) return;

			if (_hostGiveCooldown.TryGetValue(actorID, out float lastTime))
				if (Time.time - lastTime < ClassBalance.ITEM_GIVE_COOLDOWN) return;

			_hostGiveCooldown[actorID] = Time.time;
			StartCoroutine(HostGiveItemsRoutine(clientHatName, info.Sender));
		}

		private IEnumerator HostGiveItemsRoutine(string hatName, Photon.Realtime.Player requestor)
		{
			var itemsToGive = PeakClassModPlugin.BuildStartItemsByHatName(hatName);
			if (itemsToGive.Count == 0) yield break;

			Character targetChar = null;
			int actorNumber = requestor.ActorNumber;

			if (_characterCache.TryGetValue(actorNumber, out Character cachedChar) && cachedChar != null)
				targetChar = cachedChar;
			else
			{
				foreach (var ch in Character.AllCharacters)
				{
					if (ch?.photonView?.Owner == requestor)
					{
						targetChar = ch;
						_characterCache[actorNumber] = ch;
						break;
					}
				}
			}

			if (targetChar?.player == null) yield break;

			HasReceivedStartItems.Add(actorNumber);

			foreach (var itemName in itemsToGive)
			{
				var item = ItemCache.Get(itemName);
				if (item != null && targetChar.player.HasEmptySlot((ushort)item.itemID))
				{
					targetChar.player.AddItem((ushort)item.itemID, null, out _);
				}
				yield return new WaitForSeconds(0.05f);
			}
		}

using UnityEngine;
using PeakClassMod.Utils;
using System;

namespace PeakClassMod.Handlers
{
	public class ClassWatcherHandler : MonoBehaviour
	{
		private Character _character;
		private string _activeClass = "None";
		private float _lastCheckTime = 0f;
		private int _notFoundCounter = 0;

		private void Start()
		{
			_character = GetComponent<Character>();
			CheckAndApplyClass(true);
		}

		private void Update()
		{
			if (_character == null || !_character.IsLocal) return;

			if (Time.time - _lastCheckTime < 0.5f) return;
			_lastCheckTime = Time.time;

			CheckAndApplyClass(false);
		}

		public void CheckAndApplyClass(bool force)
		{
			if (_character == null || !_character.IsLocal) return;

			if (HatUtils.TryGetHatName(_character, out string hatName))
			{
				_notFoundCounter = 0;
				string targetClass = DetermineClassFromHat(hatName);
				if (force || targetClass != _activeClass)
				{
					_activeClass = targetClass;
					ApplyClass(targetClass);
				}
			}
			else
			{
				// Anti-rebond : attend 3 échecs consécutifs avant de désactiver
				_notFoundCounter++;
				if (_notFoundCounter >= 3 && _activeClass != "None")
				{
					_activeClass = "None";
					ClassManager.DeactivateAllClasses();
					PeakClassModPlugin.Log?.LogInfo("[ClassMod] Chapeau retiré -> Classes désactivées.");
				}
			}
		}

		private string DetermineClassFromHat(string hatName)
		{
			if (string.IsNullOrEmpty(hatName)) return "None";

			if (hatName.Contains("Scoutmaster")) return "Scout";
			if (hatName.Contains("Medic")) return "Medic";
			if (hatName.Contains("Crab")) return "Tech-Priest";
			if (hatName.Contains("Mushroom")) return "Mushroomancer";
			if (hatName.Contains("Propeller")) return "GreatEpee";
			if (hatName.Contains("NinjaHeadband")) return "Rogue";
			if (hatName.Contains("Bandana") || hatName.Contains("Captain")) return "General";
			if (hatName.Contains("Astronaut")) return "Astronaut";
			if (hatName.Contains("Midsummer")) return "Gobelin";
			if (hatName.Contains("WolfEars")) return "Wolf";

			return "None";
		}

		private void ApplyClass(string className)
		{
			PeakClassModPlugin.Log?.LogInfo($"[ClassMod] >>> Activation de la classe : {className} <<<");

			switch (className)
			{
				case "Scout": ClassManager.ActivateScoutClass(); break;
				case "Medic":
					ClassManager.ActivateMedicClass();
					float cMedic = _character.refs?.afflictions?.GetCurrentStatus(CharacterAfflictions.STATUSTYPE.Curse) ?? 0f;
					if (cMedic < ClassBalance.MEDIC_CURSE) _character.refs?.afflictions?.SetStatus(CharacterAfflictions.STATUSTYPE.Curse, ClassBalance.MEDIC_CURSE);
					break;
				case "Tech-Priest": ClassManager.ActivateTechPriestClass(); break;
				case "Mushroomancer":
					ClassManager.ActivateMushroomancerClass();
					ClassManager.StartMushroomancerWatch(_character);
					break;
				case "GreatEpee":
					ClassManager.ActivateGreatEpeeClass();
					ClassManager.StartGreatEpeeRoutines(_character);
					break;
				case "Rogue": ClassManager.ActivateRogueClass(); break;
				case "General":
					ClassManager.ActivateGeneralClass();
					float cGen = _character.refs?.afflictions?.GetCurrentStatus(CharacterAfflictions.STATUSTYPE.Curse) ?? 0f;
					if (cGen < ClassBalance.GENERAL_CURSE_MIN) _character.refs?.afflictions?.SetStatus(CharacterAfflictions.STATUSTYPE.Curse, ClassBalance.GENERAL_CURSE_MIN);
					break;
				case "Astronaut": ClassManager.ActivateAstronautClass(); break;
				case "Gobelin": ClassManager.ActivateGobelinClass(); break;
				case "Wolf": ClassManager.ActivateWolfClass(); break;
				default:
					ClassManager.DeactivateAllClasses();
					break;
			}

			try
			{
				if (GUIManager.instance?.bar != null)
				{
					GUIManager.instance.bar.ChangeBar();
				}
			}
			catch { }
		}
	}
}

        private void Start() { 
            _character = GetComponent<Character>(); 
            _network = GetComponent<ClassModNetwork>(); 
            if (_character?.refs != null) 
                _afflictions = _character.refs.afflictions; 
        }
        private void Update() {

            if (ClassManager.IsMedicActive()) {

            } else if (_healsRemaining < ClassBalance.MEDIC_MAX_HEALS)

                if (_network?.photonView != null) _network.photonView.RPC("ShowMedicHealEffectRPC", Photon.Pun.RpcTarget.All, _healsRemaining);

// ==================== FILE: F:\ModPeak2026\PeakModMaj\Mod_Peak_2026\src\Patches\General_AddStatus_Patch.cs ====================
using HarmonyLib;
using PeakClassMod.Utils;
using UnityEngine;
using System;

namespace PeakClassMod.Patches
{
	[HarmonyPatch(typeof(CharacterAfflictions))]
	public static class General_AddStatus_Patch
	{
		[HarmonyPatch(nameof(CharacterAfflictions.AddStatus), new Type[] {
			typeof(CharacterAfflictions.STATUSTYPE),
			typeof(float),
			typeof(bool),
			typeof(bool),
			typeof(bool)
		})]
		[HarmonyPrefix]
		public static bool Prefix(CharacterAfflictions __instance, CharacterAfflictions.STATUSTYPE statusType, ref float amount)
		{
			var ch = __instance?.character;
			if (ch == null) return true;

			// 1. FAIM PAR CLASSE
			if (ch.IsLocal && statusType == CharacterAfflictions.STATUSTYPE.Hunger)
			{
				if (ClassManager.IsAstronautActive()) amount *= ClassBalance.ASTRONAUT_HUNGER_MULT;
				else if (ClassManager.IsScoutActive()) amount *= ClassBalance.SCOUT_HUNGER_MULT;
				else if (ClassManager.IsWolfActive() && !DaySegmentUtils.IsDay()) amount *= ClassBalance.WOLF_HUNGER_MULT;
				else if (ClassManager.IsTechPriestActive()) amount *= ClassBalance.TECH_PRIEST_HUNGER_MULT;
				else if (ClassManager.IsGeneralActive()) amount *= ClassBalance.GENERAL_HUNGER_MULT;
			}

			// 2. GOBELIN : Immunité Spores
			if (ClassCheckUtil.IsGobelin(ch) && statusType == CharacterAfflictions.STATUSTYPE.Spores)
			{
				amount = 0f;
				return false;
			}

			// 3. TECH-PRIEST : Immunité Poids
			if (ClassCheckUtil.IsTechPriest(ch) && ClassManager.IsTechPriestActive())
			{
				if (statusType == CharacterAfflictions.STATUSTYPE.Weight) { amount = 0f; return false; }
			}

			// 4. MUSHROOMANCER : Immunité Poison
			if (statusType == CharacterAfflictions.STATUSTYPE.Poison && ClassCheckUtil.IsMushroom(ch))
			{
				amount = 0f;
				return false;
			}

			// 5. GENERAL : Maintien de la malédiction minimum (35%)
			if (ch.IsLocal && ClassManager.IsGeneralActive() && statusType == CharacterAfflictions.STATUSTYPE.Curse && amount < 0f)
			{
				float current = __instance.GetCurrentStatus(CharacterAfflictions.STATUSTYPE.Curse);
				if (current + amount < ClassBalance.GENERAL_CURSE_MIN)
				{
					float allowed = ClassBalance.GENERAL_CURSE_MIN - current;
					if (allowed >= 0f) return false;
					amount = allowed;
				}
			}

			return true;
		}
	}
}

    [HarmonyPatch(typeof(Item), "Update")]
    public class GobelinBlowgunEnforcer
    {
        static void Postfix(Item __instance)
        {
            // Sécurité de base
            if (__instance == null) return;

            // On ne s'intéresse qu'aux sarbacanes
            if (__instance.name.IndexOf(ClassBalance.BLOWGUN, System.StringComparison.OrdinalIgnoreCase) >= 0)
            {
                Character holder = __instance.holderCharacter;
                bool heldByGobelin = holder != null && ClassCheckUtil.IsGobelin(holder);

                // --- CAS 1 : TENU PAR UN GOBELIN ---
                if (heldByGobelin)
                {
                    // A. Logique Infini (Seulement Local pour éviter conflits réseau)
                    if (holder.IsLocal)
                    {
                        var trav = Traverse.Create(__instance);
                        var field = trav.Field("destroyOnUse");
                        // On force la NON-destruction
                        if (field.FieldExists() && field.GetValue<bool>() == true)
                        {
                            field.SetValue(false);
                        }

                        // On force les munitions à -1 (infini)
                        try
                        {
                            if (__instance.data != null && __instance.HasData(DataEntryKey.ItemUses))
                            {
                                var usesData = __instance.GetData<OptionableIntItemData>(DataEntryKey.ItemUses);
                                if (usesData != null && usesData.Value != -1) usesData.Value = -1;
                            }
                        }
                        catch { }
                    }

                    // B. Logique Soulbound (Restrictions d'inventaire)
                    if (__instance.UIData != null)
                    {
                        __instance.UIData.canDrop = false;        // Touche G interdite
                        __instance.UIData.canBackpack = false;    // Interdit de mettre dans le sac (pour donner)
                        __instance.UIData.canThrow = false;       // Interdit de lancer


                        // CORRECTION CRITIQUE ICI :
                        __instance.UIData.canPocket = true;       // DOIT ÊTRE TRUE sinon l'objet tombe quand on l'équipe !
                    }
                }
                // --- CAS 2 : TENU PAR QUELQU'UN D'AUTRE (OU AU SOL) ---
                else
                {
                    // Si ce n'est pas un Gobelin qui la tient, on remet tout à la normale.
                    // C'est crucial si le Gobelin meurt et lâche son arme.

                    // 1. Réactiver les mouvements d'inventaire
                    if (__instance.UIData != null)
                    {
                        __instance.UIData.canDrop = true;
                        __instance.UIData.canBackpack = true;
                        __instance.UIData.canThrow = true;

                        __instance.UIData.canPocket = true;
                    }

                    // 2. RETIRER L'INFINI (Correction du bug pour les autres joueurs)
                    // On remet la destruction active
                    var trav = Traverse.Create(__instance);
                    var field = trav.Field("destroyOnUse");
                    if (field.FieldExists())
                    {
                        // On remet à TRUE (l'objet doit se détruire après usage pour les non-Gobelins)
                        field.SetValue(true);
                    }

                    // On remet les munitions à 1 (usage unique) si elles étaient à -1
                    try
                    {
                        if (__instance.data != null && __instance.HasData(DataEntryKey.ItemUses))
                        {
                            var usesData = __instance.GetData<OptionableIntItemData>(DataEntryKey.ItemUses);
                            if (usesData != null && usesData.Value == -1)
                            {
                                usesData.Value = 1;
                            }
                        }
                    }
                    catch { }
                }
            }
        }

using PeakClassMod;
using PeakClassMod.Utils;
using System;
using UnityEngine;
using HarmonyLib;

namespace PeakClassMod.Patches
{
	[HarmonyPatch(typeof(CharacterAfflictions))]
	public static class UpdateStatus_TechPriest_Patch
	{
		[HarmonyPatch("GetCurrentStatus")]
		[HarmonyPostfix]
		public static void GetCurrentStatusPostfix(CharacterAfflictions __instance, CharacterAfflictions.STATUSTYPE statusType, ref float __result)
		{
			if (__instance?.character == null) return;

			// 1. GESTION DU POIDS EN TEMPS RÉEL
			if (statusType == CharacterAfflictions.STATUSTYPE.Weight)
			{
				if (ClassManager.IsTechPriestActive() && ClassCheckUtil.IsTechPriest(__instance.character))
				{
					__result = 0f;
				}
				else if (ClassManager.IsScoutActive() && ClassCheckUtil.IsScout(__instance.character))
				{
					__result *= ClassBalance.SCOUT_WEIGHT_MULT; // -50% poids
				}
				else if (ClassManager.IsGeneralActive() && ClassCheckUtil.IsGeneral(__instance.character))
				{
					__result *= ClassBalance.GENERAL_WEIGHT_MULT; // +100% poids
				}
			}

			// 2. SOCLE DE MALÉDICTION GARANTI
			if (statusType == CharacterAfflictions.STATUSTYPE.Curse)
			{
				if (ClassManager.IsGeneralActive() && ClassCheckUtil.IsGeneral(__instance.character))
				{
					if (__result < ClassBalance.GENERAL_CURSE_MIN) __result = ClassBalance.GENERAL_CURSE_MIN;
				}
				else if (ClassManager.IsMedicActive() && ClassCheckUtil.IsMedicHat(__instance.character))
				{
					if (__result < ClassBalance.MEDIC_CURSE) __result = ClassBalance.MEDIC_CURSE;
				}
			}
		}

		[HarmonyPatch("UpdateNormalStatuses")]
		[HarmonyPostfix]
		public static void UpdateNormalStatusesPostfix(CharacterAfflictions __instance)
		{
			if (__instance?.character != null && ClassCheckUtil.IsTechPriest(__instance.character) && ClassManager.IsTechPriestActive())
			{
				int weightIndex = (int)CharacterAfflictions.STATUSTYPE.Weight;
				if (__instance.currentStatuses != null && __instance.currentStatuses.Length > weightIndex)
				{
					__instance.currentStatuses[weightIndex] = 0f;
				}
			}
		}
	}
}

using System;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using Zorro.Core;
using Photon.Pun;

namespace PeakClassMod.Utils
{
	public static class HatUtils
	{
		private static string _lastLocalHatName = "None";

		private static readonly string[] KnownHats = new[]
		{
			"Hat_Scoutmaster", "Hat_Medic", "MedicHelmet", "Hat_Crab",
			"Hat_Mushroom", "Hat_Propeller", "Hat_NinjaHeadband",
			"Hat_Bandana", "Hat_Captain", "Hat_Astronaut", "Hat_Midsummer", "Hat_WolfEars"
		};

		public static void SetLastLocalHatName(string hatName)
		{
			if (!string.IsNullOrEmpty(hatName) && hatName != "Unknown" && hatName != "None")
			{
				_lastLocalHatName = hatName;
			}
		}

		public static bool TryGetHatName(Character ch, out string hatName)
		{
			hatName = null;
			if (ch == null) return false;

			// --- SOURCE 1 : Inspection directe des GameObjects attachés sur le personnage ---
			try
			{
				var transforms = ch.GetComponentsInChildren<Transform>(true);
				foreach (var t in transforms)
				{
					if (t == null || !t.gameObject.activeInHierarchy) continue;
					string objName = t.name;
					foreach (var known in KnownHats)
					{
						if (objName.IndexOf(known, StringComparison.OrdinalIgnoreCase) >= 0)
						{
							hatName = known;
							if (ch.IsLocal) SetLastLocalHatName(hatName);
							return true;
						}
					}
				}
			}
			catch { }

			// --- SOURCE 2 : PersistentPlayerDataService & Customization ---
			try
			{
				var ppds = GameHandler.GetService<PersistentPlayerDataService>();
				PersistentPlayerData data = null;
				if (ch.photonView?.Owner != null) data = ppds?.GetPlayerData(ch.photonView.Owner);
				if (data == null && ch.IsLocal && PhotonNetwork.LocalPlayer != null) data = ppds?.GetPlayerData(PhotonNetwork.LocalPlayer);

				if (data != null && Singleton<Customization>.Instance != null)
				{
					var hats = Singleton<Customization>.Instance.hats;
					int i = data.customizationData.currentHat;
					if (hats != null && i >= 0 && i < hats.Length && hats[i] != null)
					{
						hatName = hats[i].name;
						if (ch.IsLocal) SetLastLocalHatName(hatName);
						return true;
					}
				}
			}
			catch { }

			// --- SOURCE 3 : Mémoire locale ---
			if (ch.IsLocal && !string.IsNullOrEmpty(_lastLocalHatName) && _lastLocalHatName != "None")
			{
				hatName = _lastLocalHatName;
				return true;
			}

			return false;
		}

		public static bool HasAnyHat(Character ch, params string[] names)
		{
			if (!TryGetHatName(ch, out var name)) return false;
			return names.Any(n => name.IndexOf(n, StringComparison.OrdinalIgnoreCase) >= 0);
		}

		public static bool HasHat(Character ch, string expected)
		{
			return TryGetHatName(ch, out var name) && name.IndexOf(expected, StringComparison.OrdinalIgnoreCase) >= 0;
		}
	}
}

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using UnityEngine;
using UnityEngine.Networking;
using System.Collections;

namespace PeakClassMod.Utils
{
    public class ImageLoader : MonoBehaviour
    {
        public static ImageLoader Instance;

        void Awake()
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
        }

        public void Load(string path, Action<Texture2D> callback)
        {
            StartCoroutine(LoadCoroutine(path, callback));
        }

        private IEnumerator LoadCoroutine(string path, Action<Texture2D> callback)
        {
            string url = "file://" + path;

            UnityWebRequest request = UnityWebRequestTexture.GetTexture(url);
            yield return request.SendWebRequest();

            if (request.result != UnityWebRequest.Result.Success)
            {
                Debug.LogError(request.error);
                callback(null);
            }
            else
            {
                callback(DownloadHandlerTexture.GetContent(request));
            }
        }
    }
}

using System;
using System.Collections.Generic;
using UnityEngine;

namespace PeakClassMod.Utils
{
	public static class ItemCache
	{
		private static Dictionary<string, Item> _byName;
		private static float _lastRefreshTime = 0f;

		public static void Refresh()
		{
			_byName = new Dictionary<string, Item>(StringComparer.OrdinalIgnoreCase);
			var allItems = Resources.FindObjectsOfTypeAll<Item>();
			foreach (var it in allItems)
			{
				if (it == null || string.IsNullOrEmpty(it.name)) continue;
				string raw = it.name;
				string clean = raw.Replace("(Clone)", "").Trim();

				if (!_byName.ContainsKey(raw)) _byName[raw] = it;
				if (!_byName.ContainsKey(clean)) _byName[clean] = it;
			}
			_lastRefreshTime = Time.time;
		}

		public static Item Get(string name)
		{
			if (string.IsNullOrEmpty(name)) return null;

			if (_byName == null || Time.time - _lastRefreshTime > ClassBalance.ITEM_CACHE_DURATION)
			{
				Refresh();
			}

			if (_byName.TryGetValue(name, out var item)) return item;

			// Recherche floue si le nom exact a un suffixe ou préfixe
			foreach (var kvp in _byName)
			{
				if (kvp.Key.IndexOf(name, StringComparison.OrdinalIgnoreCase) >= 0 ||
					name.IndexOf(kvp.Key, StringComparison.OrdinalIgnoreCase) >= 0)
				{
					return kvp.Value;
				}
			}

			return null;
		}
	}
}

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using UnityEngine;
using System.Threading.Tasks;

namespace PeakClassMod.Utils
{
    internal static class TechPriestBlocker
    {
        private static readonly HashSet<string> BlockedTypeNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase) { "heal", "cure", "medicine", "restorehealth" };
        private static readonly HashSet<string> BlockedItemNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase) { "bandage", "firstaid", "medkit", "cure", "marshmallow" };
        private static Dictionary<int, bool> _blockCache = new Dictionary<int, bool>();
        public static bool ShouldBlockItem(Item item)
        {
            if (item == null) return false; int id = item.GetInstanceID(); if (_blockCache.TryGetValue(id, out bool cached)) return cached;
            bool block = false;
            if (item.GetComponent<Action_RestoreHunger>() != null || item.GetComponent<Action_GiveExtraStamina>() != null) block = true;
            if (!block) { string n = item.name?.Replace("(Clone)", "").Trim() ?? ""; foreach (var b in BlockedItemNames) if (n.IndexOf(b, StringComparison.OrdinalIgnoreCase) >= 0) { block = true; break; } }
            if (!block) { foreach (var c in item.GetComponents<MonoBehaviour>()) { if (c == null) continue; string t = c.GetType().Name; foreach (var b in BlockedTypeNames) if (t.IndexOf(b, StringComparison.OrdinalIgnoreCase) >= 0) { block = true; break; } if (block) break; } }
            _blockCache[id] = block; return block;
        }
        public static void ClearCache() => _blockCache.Clear();
    }
}
